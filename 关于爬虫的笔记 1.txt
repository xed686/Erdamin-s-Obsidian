# str通过encode(编码)方式转换为bytes 将人看的懂的变成电脑看不懂的
str_code = 'abc'
print(type(str_code))
byte_code = str_code.encode()
print(type(byte_code))

#bytes通过decode(解码)转化为str      将电脑看不懂的变成人看得懂的
byte_code = b'abc'
print(type(byte_code))
str_code = byte_code.decode()  #括号里面要写入要解码编码格式 如byte_code.decode("UTF-8")
print(type(str_code))
#如果要解码的不是文本而是图片的话 那么怎么转都无法转换为看得懂的文字的

重试retrying
使用超时参数能够加快整体的运行速度 但是在生活中我们使用浏览器访问网页的时候,如果发生速度很慢的情况 一般是选择重新刷新页面 代码实现如下
retrying
使用方式:安装 1 pip install retrying
            2 添加装饰器
import time
from retrying import retry

num = 1
#装饰器 形如一个@后面一个函数的格式
@retry(stop_max_attempt_number=3)#当出错了三次后就抛出异常 及最大尝试次数
def test():
    global num
    print("num=", num)
    num += 1
    time.sleep(1)
    for i in range(0,100):
        print("i", i)

if __name__ == '__main__':
    try:
        test()
    except Exception as ret:
        print("产生异常...")
        print(ret)
    else:
        print("没有异常")


session
# 在爬取某些页面的时候，服务器会需要 cookie，要得到 cookie 就需要先访问某个 URL 进行登录。
# 服务器接收到请求之后，验证用户名以及密码在登录成功的情况下会返回一个响应，这个响应的 header 中一般会有一个 set-cookie 的信息，它对应的值就是要设置的 cookie 信息。
#当你带了相关的Cookie信息后 服务器就会认为你是一个已经登陆了的用户 便会提供给你登入用户所能获得到的信息
# 每次在下一个请求中都要先获取设置的 cookie，这个过程较为麻烦，怎么解决？
# 解决方案：session（保存 cookie，下一次请求会自动带上前一次的 cookie）。
# 使用方式：
# 1. 创建一个session实例对象
s = requests.Session()

# 2. 使用上一步创建的对象发起请求
r = s.get(url1, headers)
r = s.get(url2, headers)

# # response.request.headers.get('cookies')和response.cookies有什么区别?
# Request（请求）：是你（客户端）发给服务器的 “消息”，包含你携带的信息（比如 Cookie、User-Agent 等）。
# Response（响应）：是服务器收到你的请求后，回传给你的 “回复”，包含服务器想让你保存的信息（比如新的 Cookie）。
response.request.headers.get('cookies')  #数据来源：从你发给服务器的请求头中读取 Cookie 字段（注意是大写 C），本质是 “你请求时带了什么 Cookie 给服务器”。   使用场景：查看你本次请求携带的 Cookie 内容，比如验证请求是否正确带上了登录态。                       返回的是字符串格式（如果请求头里没有 Cookie，返回 None）
response.cookies                         #数据来源：从服务器返回的响应中解析 Set-Cookie 响应头（服务器要求你保存的 Cookie），本质是 “服务器想让你存什么 Cookie”。   使用场景：获取服务器下发的新 Cookie（比如登录后服务器返回的 sessionid），便于后续请求携带。 返回的是类字典对象 可以像字典一样通过 key 获取 value，还包含 Cookie 的过期时间、域名、路径等元信息。
